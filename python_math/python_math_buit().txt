# 1. Basic mathematical functions
# Function	Meaning	Example	Output
# math.sqrt(x)	Square root	math.sqrt(64)	8.0
# math.pow(x, y)	Power	math.pow(2, 3)	8.0
# math.fabs(x)	Absolute value as float	math.fabs(-15)	15.0
# math.factorial(x)	Factorial	math.factorial(5)	120
# math.fmod(x, y)	Floating-point remainder	math.fmod(10, 3)	1.0
# math.remainder(x, y)	IEEE remainder	math.remainder(10, 3)	1.0
# math.trunc(x)	Remove decimal part	math.trunc(5.9)	5
# math.isqrt(x)	Integer square root	math.isqrt(20)	4

# 2. Rounding functions
# Function	Meaning	Example	Output
# math.ceil(x)	Round upward	math.ceil(4.2)	5
# math.floor(x)	Round downward	math.floor(4.8)	4
# round(x)	Round to nearest (built-in)	round(4.6)	5
# math.trunc(x)	Remove decimal part	math.trunc(4.8)	4

# 3. Maximum, minimum and comparison
# These are Python built-in functions, not part of the math module.
# Function	Meaning	Example	Output
# max()	Largest value	max(10, 20, 5)	20
# min()	Smallest value	min(10, 20, 5)	5
# sum()	Total	sum([10,20,30])	60
# abs()	Absolute value	abs(-50)	50
# divmod()	Quotient and remainder	divmod(10,3)	(3, 1)

# 4. Logarithmic and exponential functions

# math.exp(x)	Calculates \(e^x\)	math.exp(2)
# math.log(x)	Natural logarithm	math.log(10)

# math.log2(x)	Log base 2	math.log2(8)
# math.expm1(x)	Calculates \(e^x - 1\)	math.expm1(1)
# math.log1p(x)	Calculates \(\log(1+x)\)	math.log1p(9)
